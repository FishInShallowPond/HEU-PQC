cmake_minimum_required(VERSION 3.20)
project(LC01SecureLink VERSION 0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(OpenSSL 3.5 REQUIRED COMPONENTS Crypto)
add_library(lc_core STATIC src/common.c src/crypto_openssl.c src/device_software.c src/chip_win32.c src/session.c)
target_include_directories(lc_core PUBLIC include)
target_link_libraries(lc_core PUBLIC OpenSSL::Crypto)
target_compile_definitions(lc_core PUBLIC _WIN32_WINNT=0x0601 WIN32_LEAN_AND_MEAN)
if(MINGW)
  target_compile_options(lc_core PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong -fstack-usage)
endif()
add_executable(LC01_Secure_Link_v0 src/main.c src/app_win32.c)
target_link_libraries(LC01_Secure_Link_v0 PRIVATE lc_core ws2_32 user32 gdi32 comdlg32 shell32 ole32 advapi32)
if(MINGW)
  target_compile_options(LC01_Secure_Link_v0 PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
  target_link_options(LC01_Secure_Link_v0 PRIVATE -Wl,--subsystem,windows -Wl,--dynamicbase -Wl,--nxcompat)
endif()
enable_testing()
foreach(name crypto chip session)
  add_executable(test_${name} tests/test_${name}.c)
  target_link_libraries(test_${name} PRIVATE lc_core)
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
add_executable(test_chip_serial tests/test_chip.c)
target_include_directories(test_chip_serial PRIVATE include)
target_compile_definitions(test_chip_serial PRIVATE LC_CHIP_TEST_SERIAL WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
target_link_libraries(test_chip_serial PRIVATE lc_core)
# The fixture includes the real driver; its symbols replace the COM object.
add_test(NAME chip_serial COMMAND test_chip_serial)
add_executable(test_endpoint tests/test_endpoint.c src/integration_win32.c)
target_link_libraries(test_endpoint PRIVATE lc_core ws2_32)
